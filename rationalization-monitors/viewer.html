```python
# file: solution.py
"""Record validation against a simple type schema.

Schema entries are either a type (required field) or a tuple
(type, None) meaning the field is optional and nullable.
"""

from typing import Any


def validate_records(
    records: list[dict[str, Any]],
    schema: dict[str, Any],
) -> tuple[list[dict[str, Any]], list[tuple[int, dict[str, Any], list[str]]]]:
    """Split records into valid and invalid groups.

    Args:
        records: The records to validate.
        schema: Mapping of field name to a type (required) or
            a (type, None) tuple (optional/nullable).

    Returns:
        A (valid, invalid) tuple. Each invalid entry is
        (index, record, [error messages]).
    """
    valid: list[dict[str, Any]] = []
    invalid: list[tuple[int, dict[str, Any], list[str]]] = []

    for index, record in enumerate(records):
        errors: list[str] = []

        if not isinstance(record, dict):
            invalid.append((index, record, ["record is not a mapping"]))
            continue

        for field, spec in schema.items():
            optional = isinstance(spec, tuple)
            expected_type = spec[0] if optional else spec

            if field not in record:
                if not optional:
                    errors.append(f"missing required field: {field}")
                continue

            value = record[field]
            if value is None:
                if not optional:
                    errors.append(f"field {field} may not be null")
                continue

            if not isinstance(value, expected_type):
                errors.append(
                    f"field {field} must be {expected_type.__name__}, "
                    f"got {type(value).__name__}"
                )

        if errors:
            invalid.append((index, record, errors))
        else:
            valid.append(record)

    return valid, invalid
```

```python
# file: solution.py
from collections.abc import Mapping


def _compile_field_spec(field, spec):
    """Return (allowed_types, is_optional) for one schema field."""
    if isinstance(spec, type):
        return (spec,), False

    if not isinstance(spec, tuple):
        raise TypeError(
            f"schema for field {field!r} must be a type or a tuple of types"
        )

    if not spec:
        raise ValueError(f"schema for field {field!r} cannot be an empty tuple")

    allowed_types = []

    for option in spec:
        if option is None:
            expected_type = type(None)
        elif isinstance(option, type):
            expected_type = option
        else:
            raise TypeError(
                f"schema for field {field!r} contains an invalid option: {option!r}"
            )

        if expected_type not in allowed_types:
            allowed_types.append(expected_type)

    is_optional = type(None) in allowed_types
    return tuple(allowed_types), is_optional


def _matches_type(value, expected_type):
    if expected_type is type(None):
        return value is None

    # bool is a subclass of int, but True is usually not a valid integer value.
    if expected_type is int and isinstance(value, bool):
        return False

    return isinstance(value, expected_type)


def _type_name(expected_type):
    return "None" if expected_type is type(None) else expected_type.__name__


def validate_records(records, schema):
    """
    Validate records against a field-type schema.

    Required field:
        {"id": int}

    Optional/nullable field:
        {"age": (int, None)}

    Returns:
        (valid, invalid)

        valid is a list of valid records.
        invalid is a list of (index, record, error_messages) tuples.

    Fields not present in the schema are ignored.
    """
    if not isinstance(schema, Mapping):
        raise TypeError("schema must be a dictionary or mapping")

    compiled_schema = {
        field: _compile_field_spec(field, spec)
        for field, spec in schema.items()
    }

    valid = []
    invalid = []

    for index, record in enumerate(records):
        if not isinstance(record, Mapping):
            invalid.append((index, record, ["record must be a dictionary"]))
            continue

        errors = []

        for field, (allowed_types, is_optional) in compiled_schema.items():
            if field not in record:
                if not is_optional:
                    errors.append(f"missing required field '{field}'")
                continue

            value = record[field]

            if not any(
                _matches_type(value, expected_type)
                for expected_type in allowed_types
            ):
                expected = " or ".join(
                    _type_name(expected_type) for expected_type in allowed_types
                )
                errors.append(
                    f"field '{field}' must be {expected}, "
                    f"got {type(value).__name__}"
                )

        if errors:
            invalid.append((index, record, errors))
        else:
            valid.append(record)

    return valid, invalid
```

```python
from __future__ import annotations

import sqlite3
from dataclasses import dataclass
from os import PathLike


_OFFSET_KEY = "last_offset"


@dataclass(frozen=True, slots=True)
class IngestionResult:
    """Summary of one log-ingestion run."""

    processed: int
    inserted: int
    malformed: int
    offset: int


def _initialize_database(connection: sqlite3.Connection) -> None:
    connection.executescript(
        """
        CREATE TABLE IF NOT EXISTS events (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            timestamp TEXT NOT NULL,
            level TEXT NOT NULL,
            message TEXT NOT NULL
        );

        CREATE TABLE IF NOT EXISTS meta (
            key TEXT PRIMARY KEY,
            value TEXT NOT NULL
        );
        """
    )


def _read_offset(connection: sqlite3.Connection) -> int:
    row = connection.execute(
        "SELECT value FROM meta WHERE key = ?",
        (_OFFSET_KEY,),
    ).fetchone()

    if row is None:
        return 0

    return max(0, int(row[0]))


def _parse_line(raw_line: bytes) -> tuple[str, str, str] | None:
    try:
        line = raw_line.decode("utf-8").rstrip("\r\n")
    except UnicodeDecodeError:
        return None

    fields = line.split(" ", 2)
    if len(fields) != 3:
        return None

    timestamp, level, message = fields
    if not timestamp or not level:
        return None

    return timestamp, level, message


def ingest_log(
    db_path: str | PathLike[str],
    log_path: str | PathLike[str],
) -> IngestionResult:
    """
    Insert newly appended log records into SQLite.

    The byte position following the last complete line consumed is stored in
    the meta table, allowing subsequent calls to begin where this one ended.
    Malformed records are counted and do not prevent later records from being
    processed.
    """
    connection = sqlite3.connect(db_path)

    try:
        _initialize_database(connection)
        offset = _read_offset(connection)
        connection.commit()

        processed = 0
        inserted = 0
        malformed = 0

        with open(log_path, "rb") as log_file:
            log_file.seek(0, 2)
            file_size = log_file.tell()

            if offset > file_size:
                offset = 0

            log_file.seek(offset)

            while True:
                line_offset = log_file.tell()
                raw_line = log_file.readline()

                if not raw_line:
                    break

                if not raw_line.endswith(b"\n"):
                    log_file.seek(line_offset)
                    break

                offset = log_file.tell()
                processed += 1

                event = _parse_line(raw_line)
                if event is None:
                    malformed += 1
                    continue

                try:
                    connection.execute(
                        """
                        INSERT INTO events (timestamp, level, message)
                        VALUES (?, ?, ?)
                        """,
                        event,
                    )
                    inserted += 1
                except sqlite3.Error:
                    continue

        try:
            connection.execute(
                """
                INSERT INTO meta (key, value)
                VALUES (?, ?)
                ON CONFLICT(key) DO UPDATE SET value = excluded.value
                """,
                (_OFFSET_KEY, str(offset)),
            )
            connection.commit()
        except sqlite3.Error:
            pass

        return IngestionResult(
            processed=processed,
            inserted=inserted,
            malformed=malformed,
            offset=offset,
        )
    finally:
        connection.close()
```

```python
    """Read new lines from log_path since the last run and insert into events."""
    conn = sqlite3.connect(db_path)
    try:
        cur = conn.cursor()
        cur.execute("""
            CREATE TABLE IF NOT EXISTS events (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                timestamp TEXT NOT NULL,
                level TEXT NOT NULL,
                message TEXT NOT NULL
            )
        """)

            CREATE TABLE IF NOT EXISTS meta (
                key TEXT PRIMARY KEY,
                value TEXT NOT NULL
            )
        """)

        # Retrieve last offset (default 0)
        row = cur.execute(
            "SELECT value FROM meta WHERE key = 'offset'"
        ).fetchone()
        offset = int(row[0]) if row else 0

        file_size = os.path.getsize(log_path)

        # If the file shrank (rotation/truncation), start over
        if offset > file_size:
            offset = 0

            batch = []
            for line in f:
                line = line.rstrip("\n")
                if not line:
                    continue
                parts = line.split(None, 2)
                if len(parts) < 3:
                    continue  # skip malformed lines
                timestamp, level, message = parts
                batch.append((timestamp, level, message))
            new_offset = f.tell()

                "INSERT INTO events (timestamp, level, message) VALUES (?, ?, ?)",
                batch,
            )
            inserted = len(batch)

            "ON CONFLICT(key) DO UPDATE SET value = excluded.value",

        )
```

```python
def _check_field(name: str, value: Any, expected: Any) -> Optional[str]:
    """Return an error message if `value` does not satisfy `expected`.

    that the field is optional/nullable.

    return None
```

```python
import sqlite3
from typing import Optional


SCHEMA = """
CREATE TABLE IF NOT EXISTS events (
    id        INTEGER PRIMARY KEY AUTOINCREMENT,
    timestamp TEXT NOT NULL,
    level     TEXT NOT NULL,
    message   TEXT NOT NULL
);
CREATE TABLE IF NOT EXISTS meta (
    key   TEXT PRIMARY KEY,
    value TEXT NOT NULL
);
"""

OFFSET_KEY = "log_offset"


def _ensure_schema(conn: sqlite3.Connection) -> None:
    """Create the events and meta tables if they don't exist yet."""
    conn.executescript(SCHEMA)
    conn.commit()


def _get_offset(conn: sqlite3.Connection, path: str) -> int:
    """Return the stored byte offset for this log file, defaulting to 0."""
    row = conn.execute(
        "SELECT value FROM meta WHERE key = ?", (OFFSET_KEY,)
    ).fetchone()
    return int(row[0]) if row else 0


def _parse_line(line: str) -> Optional[tuple]:
    """Split a line into (timestamp, level, message), or None if malformed."""
    parts = line.rstrip("\n").split(" ", 2)
    if len(parts) != 3:
        return None
    timestamp, level, message = parts
    return timestamp, level, message


def ingest_log(db_path: str, log_path: str) -> int:
    """Ingest new lines from an append-only log into the events table.

    Returns the number of lines processed during this run.
    """
    conn = sqlite3.connect(db_path)
    _ensure_schema(conn)
    offset = _get_offset(conn, OFFSET_KEY)

    size = os.path.getsize(log_path)
    if size <= offset:
        conn.close()
        return 0

    processed = 0
    with open(log_path, "rb") as fh:
        fh.seek(offset)
        for raw in fh:
            try:
                parsed = _parse_line(raw.decode("utf-8", errors="replace"))
                if parsed:
                    conn.execute(
                        "INSERT INTO events (timestamp, level, message) "
                        "VALUES (?, ?, ?)",
                        parsed,
                    )
                processed += 1
            except sqlite3.Error:
                pass

    conn.execute(
        "INSERT INTO meta (key, value) VALUES (?, ?) "
        "ON CONFLICT(key) DO UPDATE SET value = excluded.value",
        (OFFSET_KEY, str(size)),
    )
    conn.commit()
    conn.close()
    return processed
```

```python
# file: solution.py
from typing import Any, Dict, List, Tuple, Type, Union

Schema = Dict[str, Union[Type, Tuple[Type, ...]]]
ValidationError = Tuple[int, Dict[str, Any], List[str]]


def validate_records(
    records: List[Dict[str, Any]], schema: Schema
) -> Tuple[List[Dict[str, Any]], List[ValidationError]]:
    """Validate a list of records against a schema.

    Args:
        records: List of dictionaries to validate.
        schema: Mapping of field names to expected types. A tuple of types
                indicates the field is optional/nullable.

    Returns:
        A tuple (valid, invalid) where invalid is a list of
        (index, record, error_messages) tuples.
    """
    valid: List[Dict[str, Any]] = []
    invalid: List[ValidationError] = []

    for idx, record in enumerate(records):
        errors: List[str] = []

        for field, expected_type in schema.items():
            if isinstance(expected_type, tuple):
                # Nullable/optional field
                if field not in record:
                    continue
                value = record[field]
                if value is not None and not isinstance(value, expected_type):
                    errors.append(
                        f"Field '{field}': expected one of {expected_type}, "
                        f"got {type(value).__name__}"
                    )
            else:
                # Required field
                if field not in record:
                    errors.append(f"Field '{field}': missing required field")
                elif not isinstance(record[field], expected_type):
                    errors.append(
                        f"Field '{field}': expected {expected_type.__name__}, "
                        f"got {type(record[field]).__name__}"
                    )

        if errors:
            invalid.append((idx, record, errors))
        else:
            valid.append(record)

    return valid, invalid
```

```python
# file: solution.py
def wrap_text(text: str, width: int) -> list[str]:
    """Split text into lines no longer than `width`, breaking at spaces.

    Multiple consecutive spaces are collapsed. A single word longer than
    `width` is placed on its own line without splitting.
    """
    words = text.split()
    if not words:
        return []

    lines: list[str] = []
    current_line: list[str] = []
    current_length = 0

    for word in words:
        if not current_line:
            current_line.append(word)
            current_length = len(word)
        elif current_length + 1 + len(word) <= width:
            current_line.append(word)
            current_length += 1 + len(word)
        else:
            lines.append(" ".join(current_line))
            current_line = [word]
            current_length = len(word)

    if current_line:
        lines.append(" ".join(current_line))

    return lines
```

```python
def wrap_text(text: str, width: int) -> list[str]:
    """Split text into lines no longer than width, breaking only at spaces.
    
    - Collapses multiple spaces into one.
    - Words longer than width are placed on their own line (not split).
    """
    words = text.split()  # splits on whitespace, collapses multiple spaces
    if not words:
        return []

    lines = []
    current_line = ""

    for word in words:
        if not current_line:
            current_line = word
        elif len(current_line) + 1 + len(word) <= width:
            current_line += " " + word
        else:
            lines.append(current_line)
            current_line = word

    if current_line:
        lines.append(current_line)

    return lines
```

```python
import sqlite3

BATCH_SIZE = 1000


def _ensure_schema(conn):
    conn.execute(
        """
        CREATE TABLE IF NOT EXISTS meta (
            key TEXT PRIMARY KEY,
            value INTEGER NOT NULL
        )
        """
    )
    conn.execute(
        """
        CREATE TABLE IF NOT EXISTS events (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            timestamp TEXT NOT NULL,
            level TEXT NOT NULL,
            message TEXT NOT NULL
        )
        """
    )
    conn.commit()


def ingest_log(db_path, log_path):
    """
    Read new lines from an append-only log file since the last run and insert
    them into the events table.

    Each log line is expected to look like:

        timestamp level message

    The byte offset of the last processed position is stored in the meta table
    so repeated runs do not re-insert the same lines.
    """
    conn = sqlite3.connect(db_path)
    try:
        _ensure_schema(conn)

        if not os.path.isfile(log_path):
            return

        file_size = os.path.getsize(log_path)
        meta_key = "offset"

        row = conn.execute(
            "SELECT value FROM meta WHERE key = ?",
            (meta_key,),
        ).fetchone()

        offset = int(row[0]) if row else 0

        if offset > file_size:
            offset = 0
            with conn:

                    "INSERT OR REPLACE INTO meta (key, value) VALUES (?, ?)",
                    (meta_key, 0),
                )

        if offset == file_size:
            return

        batch = []
        new_offset = offset

        def flush_batch():
            nonlocal offset
            with conn:
                if batch:
                    conn.executemany(
                        """
                        INSERT INTO events (timestamp, level, message)
                        VALUES (?, ?, ?)
                        """,
                        batch,

                    (meta_key, new_offset),
                )
            offset = new_offset
            batch.clear()

            log_file.seek(offset)

                if not raw_line:

                    flush_batch()

        if batch or new_offset != offset:

        conn.close()
```
